核心代码 · fetch 请求流程
fetch("https://dummyjson.com/products?limit=8")
  .then(res => {
    if (!res.ok) throw new Error("HTTP " + res.status);
    return res.json();               // 把响应体解析为 JSON 对象
  })
  .then(data => render(data.products)) // 拿到商品数组并渲染
  .catch(err => showError(err));       // 统一处理异常
