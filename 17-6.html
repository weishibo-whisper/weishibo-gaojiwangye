    dumplings.js 
    // 用 setTimeout 模拟一个耗时任务
const wait = ms => new Promise(resolve =>
  setTimeout(resolve, ms)
);

async function cookDumplings() {
  await wait(6000); // 买原料
  await wait(2000); // 取原料

  // 两个任务同时开始，都完成才继续
  await Promise.all([
    wait(5000), // 和面
    wait(4000)  // 和馅
  ]);

  await wait(6000); // 包饺子
  await wait(3000); // 吃饺子
}

    wait() 返回 Promise；await 等待它兑现； Promise.all 会等数组里的任务全部完成后再继续。

    关键点：和面 5 秒、和馅 4 秒如果依次执行需要 9 秒；放进 Promise.all 后同时开始， 只需等待较慢的和面完成（5 秒）。整条流程约 6 + 2 + 5 + 6 + 3 = 22 秒。
